cmake_minimum_required(VERSION 3.20)
project(linux_ngx_probe LANGUAGES CXX)
find_package(Vulkan REQUIRED)
set(NGX_SDK_ROOT "" CACHE PATH "NGX SDK staging directory with include/ and lib/")
find_path(NGX_INCLUDE nvsdk_ngx_vk.h PATHS "${NGX_SDK_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(NGX_LIBRARY nvsdk_ngx PATHS "${NGX_SDK_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
add_executable(linux-ngx-probe main.cpp)
target_compile_features(linux-ngx-probe PRIVATE cxx_std_20)
target_include_directories(linux-ngx-probe PRIVATE "${NGX_INCLUDE}")
target_link_libraries(linux-ngx-probe PRIVATE Vulkan::Vulkan "${NGX_LIBRARY}" ${CMAKE_DL_LIBS})
